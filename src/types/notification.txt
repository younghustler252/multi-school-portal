import { Types } from "mongoose";
import { UserRole } from "./UserRole";

/* ============================
   Notification Types
=========================== */
export enum NotificationType {
    INFO = "info",
    WARNING = "warning",
    ALERT = "alert",
    EXAM = "exam",
    ANNOUNCEMENT = "announcement"
}

/* ============================
   Notification Sender Types
=========================== */
export enum NotificationSender {
    SYSTEM = "system",   // System (platform admin or super admin)
    USER = "user"       // User (school admin, teacher, student, etc.)
}

/* ============================
   Notification Interface
=========================== */
export interface INotification {
    tenantId: Types.ObjectId;             // School ID
    recipientId: Types.ObjectId;          // User ID
    recipientRole: UserRole;               // Role of the recipient (e.g., student, teacher)
    title: string;                         // Notification title
    message: string;                       // Full message of the notification
    type: NotificationType;                // Type of notification (info, warning, etc.)
    isRead: boolean;                       // Whether the notification has been read
    createdAt: Date;                       // Timestamp of creation
    sender: NotificationSender;            // Who sent the notification (system or user)
    meta?: Record<string, any>;            // Optional additional data (e.g., exam ID, assignment ID)
}
